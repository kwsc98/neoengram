声明式策略 · kitchen-raw

可按制品单独配置 + 集群默认 · 生命周期控制器自动执行 · 可观测/可暂停/可预演 
artifact: kitchen-raw

keep_recent: 5                              # 所有存储位置保留最近 5 版字节
archive: { to: oss-archive, when: aged }    # aged = 超出 keep_recent 时先归档
on_commit: { sync_to: [center-nas] }        # 新版本提交后自动同步到中心 NAS
playground_gc: ∞                            # 默认永不回收

stores:                                     # 存储位置级覆盖
  center-nas:
    keep_recent: 3
    archive: false                          # 中心是暂存，释放直接删（不归档）
  edge-b-nas:
    keep_recent: 2                          # 跨边缘副本只留 2 版
